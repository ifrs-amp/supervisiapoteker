📖 Cara Menghubungkan ke Google Sheets

  1. Buka Google Sheets → buat spreadsheet baru, beri nama misal "Riwayat Supervisi Apoteker".
  2. Menu Extensions → Apps Script.
  3. Hapus kode yang ada, lalu paste kode di bawah ini.
  4. Klik Deploy → New deployment → Web app.
  5. Set Execute as: Me dan Who has access: Anyone. Klik Deploy & izinkan akses.
  6. Copy Web app URL (berakhiran /exec), paste ke kolom URL di halaman Pengaturan, lalu klik Simpan URL.
function doPost(e) {
  const SHEET_NAME = 'Riwayat';
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) sh = ss.insertSheet(SHEET_NAME);

  const data = JSON.parse(e.postData.contents);

  if (data.action === 'syncAll') {
    sh.clear();
    sh.appendRow(['No','Nama Apoteker','Minggu','Ruangan','Tanggal','Waktu Input']);
    data.rows.forEach(function(r, i){
      sh.appendRow([i+1, r.nama, r.minggu, r.ruangan, r.tanggal || '-', r.createdAt]);
    });
  } else {
    if (sh.getLastRow() === 0) {
      sh.appendRow(['No','Nama Apoteker','Minggu','Ruangan','Tanggal','Waktu Input']);
    }
    sh.appendRow([sh.getLastRow(), data.nama, data.minggu, data.ruangan, data.tanggal || '-', data.createdAt]);
  }

  sh.setColumnWidth(2, 180);
  sh.setColumnWidth(4, 180);

  return ContentService
    .createTextOutput(JSON.stringify({status:'ok'}))
    .setMimeType(ContentService.MimeType.JSON);
}
